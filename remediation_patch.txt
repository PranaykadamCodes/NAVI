# This script upgrades Django to a version that addresses the SQL injection vulnerability.
# It assumes that pip is used for package management.

# Upgrade Django to the latest version in the 5.2 series (>= 5.2.9)
pip install "django>=5.2.9,<5.3"

# OR Upgrade Django to the latest version in the 5.1 series (>= 5.1.15)
# pip install "django>=5.1.15,<5.2"

# OR Upgrade Django to the latest version in the 4.2 series (>= 4.2.27)
# pip install "django>=4.2.27,<4.3"

# Verify the installed Django version
django-admin --version
